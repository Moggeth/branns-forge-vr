```powershell
& .\tools\godot\Godot_v4.7.2-stable_win64_console.exe --path godot --xr-mode off --rendering-method mobile --resolution 1000x800 --audio-driver Dummy --quit-after 120 --script res://fairy_art_capture.gd -- --capture-output=<absolute-output-folder>
```
